namespace = BA_ark_start

# 原版初始化完成后，把开局模块替换为 BA 专用模块。
country_event = {
	id = BA_ark_start.1
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_origin = origin_Kivotos_Unnamed_starting
		is_nomadic = yes
	}
	immediate = {
		# 与旧起源一致：开局解锁什亭之匣并接通 Arona 通讯。
		give_technology = {
			tech = tech_BA_Shittem_starting
			message = no
		}
		set_country_flag = BA_Arona_Established
		country_event = { id = BA_ark_start.3 }
		if = {
			limit = { exists = event_target:BA_Shittem_country }
			event_target:BA_Shittem_country = {
				establish_communications_no_message = root
			}
		}
		if = {
			limit = { exists = capital_scope }
			capital_scope = {
				# 三个基座区各一个，不预设任何区域特化。
				remove_all_districts = yes
				add_district = BA_ark_district_slot_01
				add_district = BA_ark_district_slot_02
				add_district = BA_ark_district_slot_03

				# 军国主义默认开局会添加要塞；方舟开局将其移除。
				remove_building = building_stronghold
				remove_building = building_fortress
				set_timed_carrier_flag = {
					flag = ignore_ai_building_limitations
					days = 1
				}
				add_building = {
					zone = zone_default
					building = building_BA_Schale_office
				}
				validate_planet_buildings_and_districts = yes
			}
		}
		country_event = { id = BA_ark_start.2 days = 1 }
	}
}

# 照搬通用的 Arona 起始对话，并接回既有 BA_Shittem.0102 后续链。
country_event = {
	id = BA_ark_start.3
	title = "BA_Shittem.0101.name"
	desc = "BA_Shittem.0101.desc"
	picture = GFX_BA_Emergencies_0004
	is_triggered_only = yes
	option = {
		name = BA_Shittem.0101.a
	}
	option = {
		name = BA_Shittem.0101.b
	}
	after = {
		hidden_effect = {
			country_event = { id = BA_Shittem.0102 }
		}
	}
}

country_event = {
	id = BA_ark_start.2
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_origin = origin_Kivotos_Unnamed_starting
		is_nomadic = yes
		exists = capital_scope
		capital_scope = {
			ship? = { is_ship_size = BA_arkship_tier_1 }
		}
	}
	immediate = {
		capital_scope = {
			ship? = {
				starbase = {
					set_starbase_module = {
						slot = 1
						module = BA_arkship_engineering_bay
					}
					set_starbase_module = {
						slot = 2
						module = BA_arkship_docking_bay
					}
				}
			}
		}
	}
}
