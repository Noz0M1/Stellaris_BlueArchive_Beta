# 触发器
country_event = {
	id = BA_rename.1040
	is_triggered_only = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		from = {
			BA_rename_identity_target_check = yes
		}
	}
	immediate = {
		from = {
			leader_event = {
				id = BA_rename.1041
			}
		}
	}
}

# 效果器（触发器 or 手动处理）
leader_event = {
	id = BA_rename.1041
	is_triggered_only = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		BA_rename_identity_target_check = yes
	}
	immediate = {
		# 赋予身份
		BA_rename_base_identity = yes
		# 名称立绘
		BA_rename_base_portrait = yes
		# 社团特质
		BA_rename_base_club_trait = yes
		# 个人特质
		BA_rename_base_skin_trait = yes
	}
}

# 5. 招募单个处理

# 招募
leader_event = {
	id = BA_rename.1050
	is_triggered_only = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		has_leader_flag = BA_rename_identity
	}
	immediate = {
		# 个人特质
		BA_rename_base_skin_trait = yes
	}
}

# 升级
country_event = {
	id = BA_rename.1051
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		from = {
			has_leader_flag = BA_rename_identity
		}
	}
	immediate = {
		# 个人特质
		from = {
			if = {
				limit = {
					NOT = {
						has_leader_flag = BA_rename_skin_trait
					}
				}
				BA_rename_base_skin_trait = yes
			}
		}
	}
}
