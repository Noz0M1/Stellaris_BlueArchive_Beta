#开关
country_event = {
	id = BA_rename.1001
	is_triggered_only = yes
	title = BA_rename.1001.title
	desc = BA_rename.1001.desc
	picture = GFX_BA_Azusa_in_Trinity
	fire_only_once = yes
	trigger = {
		is_country_type = default
		is_ai = no
		any_country = {
			has_authority = auth_kivotos
		}
	}
	option = {
		name = BA_rename.1001.on
		custom_tooltip = BA_rename.1001.on.tip
		default_hide_option = yes
		hidden_effect = {
			set_global_flag = BA_leader_rename_activated
			set_global_flag = BA_leader_rename_add_club_trait
			set_global_flag = BA_leader_rename_add_skin_trait
		}
	}
	option = {
		name = BA_rename.1001.partial
		custom_tooltip = BA_rename.1001.partial.tip
		hidden_effect = {
			set_global_flag = BA_leader_rename_activated
		}
	}
	option = {
		name = BA_rename.1001.off
	}
	after = {
		hidden_effect = {
			every_country = {
				limit = {
					is_country_type = default
				}
				country_event = {
					id = BA_rename.1002
				}
			}
		}
	}
}

country_event = {
	id = BA_rename.1004
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		from = {
			species = {
				BA_has_halo_organic = yes
			}
		}
	}
	immediate = {
		from = {
			#检查命名标志
			BA_check_rename_flag = yes

			BA_rename_leader = yes
		}
	}
}

#招募时处理领袖，尤其是初始3级以上的领袖
country_event = {
	id = BA_rename.1005
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_global_flag = BA_leader_rename_add_skin_trait
		from = {
			species = {
				BA_has_halo_organic = yes
			}
			BA_rename_special_check_1 = yes
		}
	}
	immediate = {
		BA_change_skin_trait = yes
	}
}

#已存在领袖修复
country_event = {
	id = BA_rename.1006
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_global_flag = BA_leader_rename_activated
		NOT = {
			has_country_flag = BA_leader_rename_modification_action
		}
	}
	immediate = {
		set_timed_country_flag = {
			flag = BA_leader_rename_modification_action
			days = 5
		}
		BA_set_rename_fix_flag = yes
		BA_iterate_rename_fix_flag = yes
	}
}
