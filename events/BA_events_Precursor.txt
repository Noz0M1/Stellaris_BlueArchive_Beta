# 初始化
country_event = {
	id = BA_Precursor.0001
	is_triggered_only = yes
	hide_window = yes
	fire_only_once = yes
	trigger = {
		is_country_type = global_event
		any_country = {
			is_ai = no
			has_authority = auth_kivotos
		}
	}
	immediate = {
		#### 设定难度！！
		set_variable = {
			which = BA_Crisis_Strength
			value = 0
		}
		get_galaxy_setup_value = {
			which = BA_Crisis_Strength
			setting = crisis_strength_scale
		}
		# <5 简单；<25 中等；<250 困难；else 疯狂
		if = {
			limit = {
				check_variable = {
					which = BA_Crisis_Strength
					value < 5
				}
			}
			set_global_flag = BA_Crisis_Normal
		}
		else_if = {
			limit = {
				check_variable = {
					which = BA_Crisis_Strength
					value < 25
				}
			}
			set_global_flag = BA_Crisis_Hardcore
		}
		else_if = {
			limit = {
				check_variable = {
					which = BA_Crisis_Strength
					value < 150
				}
			}
			set_global_flag = BA_Crisis_Insane
		}
		else_if = {
			limit = {
				check_variable = {
					which = BA_Crisis_Strength
					value < 250
				}
			}
			set_global_flag = BA_Crisis_Torment
		}
		else = {
			set_global_flag = BA_Crisis_Lunatic
		}
		#### 创建国家
		create_species = {
			name = "NAME_BA_Precursor_species"
			class = Schale
			portrait = Schale_pr
			traits = {
				trait = trait_pc_ringworld_habitable_preference
				trait = trait_Halo_precursor
				trait = trait_BA_top_notchism
				trait = trait_BA_starlink
				trait = trait_BA_gammer
			}
			allow_negative_traits = no
			effect = {
				save_global_event_target_as = BA_Precursor_species
			}
		}
		create_country = {
			name = NAME_BA_Precursor_country
			species = event_target:BA_Precursor_species
			type = Moe_Xcom
			ignore_initial_colony_error = yes
			authority = auth_kivotos_pr
			ethos = {
				ethic = ethic_fanatic_egalitarian
				ethic = ethic_pacifist
			}
			origin = origin_ba_fallen_empire
			civics = {
				civic = civic_ba_fe
			}
			flag = {
				icon = {
					category = "special"
					file = "flag_MX.dds"
				}
				background = {
					category = "backgrounds"
					file = "flag_BG_31.dds"
				}
				colors = {
					"satin_burgundy"
					"intense_blue"
					"null"
					"null"
				}
			}
			effect = {
				set_country_flag = protected_from_queen_storm #合成女王天灾保护
				##########
				save_global_event_target_as = BA_Precursor_country
				set_country_flag = BA_Precursor
				##########
				create_leader = {
					class = official
					species = event_target:BA_Precursor_species
					gender = female
					name = NAME_Seia
					skill = 10
					set_age = 17
					immortal = yes
					event_leader = yes
					traits = {
						trait = leader_trait_councilor_fe
					}
					effect = {
						change_leader_portrait = BA_leader_Seia_pr
						owner = {
							assign_leader = prev
						}
					}
				}
				##########
				give_technology = {
					tech = tech_BA_fe
					message = no
				}
				add_resource = {
					minerals = 5000
					energy = 5000
					food = 5000
					alloys = 5000
					consumer_goods = 5000
					influence = 800
					unity = 500000
					#sr_dark_matter = 100000
					sr_keystone = 500000
					sr_pyroxenes = 500000
				}
				add_global_ship_design = NAME_BA_FE_MISSILE_SHIP
				add_global_ship_design = NAME_BA_FE_ARTILLERY_SHIP
				add_global_ship_design = NAME_BA_FE_SWARM_SHIP
				add_global_ship_design = NAME_BA_FE_HUB_SHIP
				add_global_ship_design = NAME_BA_FE_DEFENCE_CANON
				add_global_ship_design = NAME_BA_FE_CONSTRUCTION
				##########
				country_event = {
					id = BA_Precursor.0100
					days = 3600
					random = 3600
				}
			}
		}
		#### 创建星系
		BA_generate_Precursor_cluster = yes
		##先驱者线
		event_target:BA_Precursor_species = {
			# save_global_event_target_as = BA_fe_species
			set_species_homeworld = event_target:BA_fe_homeworld
		}
		event_target:BA_Precursor_country = {
			# save_global_event_target_as = BA_fe_country
			##########
			# BA_fe_fleet_initial = yes
		}
	}
}

	custom_gui_option = BA_Precursor_Reward_option
	title = BA_Precursor.coop_reward
	desc = BA_Precursor.coop_reward.desc
	# 退出兜底：当所有奖励均已领取时，保证事件仍有有效选项。
	option = {
		name = BA_Precursor.exit
		default_hide_option = yes
		hidden_effect = {
			country_event = {
				id = BA_Precursor.0152
			}
		}
	}
	# 1档：资源
	option = {
		name = BA_Precursor.coop_reward.1
		custom_tooltip = BA_Precursor.coop_reward.1.tip
		is_dialog_only = yes
		response_text = BA_Precursor.coop_reward.1.re
		trigger = {
			check_variable = {
				which = BA_Precursor_coop_claimed
				value < @BA_Precursor_coop_reward_1
			}
		}
		allow = {
			# 一档不检测合作量，直接领取
			custom_tooltip = {
				fail_text = BA_Precursor.coop_reward.nochoice
				OR = {
					has_country_flag = BA_Precursor_reward_choice_base
					has_country_flag = BA_Precursor_reward_choice_tech
					has_country_flag = BA_Precursor_reward_choice_industry
					has_country_flag = BA_Precursor_reward_choice_trade
				}
			}
		}
		BA_Precursor_reward_resource = yes
		hidden_effect = {
			set_variable = {
				which = BA_Precursor_coop_claimed
				value = @BA_Precursor_coop_reward_1
			}
		}
	}
	# 2档：开放“天使24”便利店
	option = {
		name = BA_Precursor.coop_reward.2
		custom_tooltip = BA_Precursor.coop_reward.2.tip
		is_dialog_only = yes
		response_text = BA_Precursor.coop_reward.2.re
		trigger = {
			check_variable = {
				which = BA_Precursor_coop_claimed
				value < @BA_Precursor_coop_reward_2
			}
		}
		allow = {
			if = {
				limit = {
					check_variable = {
						which = BA_Precursor_coop_score
						value < @BA_Precursor_coop_reward_2
					}
				}
				custom_tooltip = {
					fail_text = BA_Precursor.coop_reward.notreach
					always = no
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = BA_Precursor_coop_claimed
						value < @BA_Precursor_coop_reward_1
					}
				}
				custom_tooltip = {
					fail_text = BA_Precursor.coop_reward.unclamed
					always = no
				}
			}
			else = {
				custom_tooltip = {
					fail_text = BA_Precursor.coop_reward.nochoice
