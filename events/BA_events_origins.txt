	id = BA_origin.0001
	hide_window = yes
	is_triggered_only = yes

		has_origin = origin_Kivotos_Eden_treaty
	}

			id = BA_origin.0002
		}

			tech = tech_BA_Shittem_starting
			message = no
		}
		set_country_flag = BA_Arona_Established
		event_target:BA_Shittem_country = {
			establish_communications_no_message = root
		}
		capital_scope = {
			clear_planet_modifiers = yes
			# 区划建筑
			validate_planet_buildings_and_districts = yes
			remove_all_districts = yes
			add_district = BA_districts_campus

			add_district = BA_districts_military_factory
			add_district = BA_districts_military_factory
			add_district = BA_districts_suburbs
			add_district = BA_districts_suburbs
			add_district = BA_districts_suburbs
			add_district = BA_districts_suburbs
			add_district = BA_districts_municipality_A
			remove_all_buildings = yes
			add_building = {
				zone = zone_default
				building = building_capital
			}
			add_building = {
				zone = zone_default
				building = building_BA_Locker_room
			}
			add_building = {
				zone = zone_default
				building = building_BA_basis_schola_1
			}
			# 区块
			clear_deposits = yes #清除原有区块
			add_deposit = d_black_market #黑市区块
			add_deposit = d_broken_Abydos_school #阿拜多斯校区
			add_deposit = d_broken_Millennium_school #千年校区
			add_deposit = d_broken_Trinity_school #三一校区

			#人口
			every_owned_pop_group = {
				kill_pop_group = {
					pop_group = this
					percentage = 0.25
				}
			}
			check_planet_employment = yes
			validate_planet_buildings_and_districts = yes
		}
	}
