		always = yes
	}
	potential = {
		always = yes

					value = -1
				}
			}

	}

#总开关
button_effect_BA_setup_leader_rename_detail = {
	allow = {
		always = yes
	}
	potential = {
		always = yes
	}
	effect = {
		custom_tooltip = Setup_leader_rename_active_tip
	}
}

button_effect_BA_setup_leader_rename_on = {
	allow = {
		always = yes
	}
	potential = {
		hidden_trigger = {
			NOT = {
				has_country_flag = BA_Setting_leader_rename_on
			}
		}
	}
	effect = {
		hidden_effect = {
			set_country_flag = BA_Setting_leader_rename_on
			set_country_flag = BA_Setting_leader_rename_add_club_trait
			set_country_flag = BA_Setting_leader_rename_add_skin_trait
		}
	}
}

button_effect_BA_setup_leader_rename_off = {
	allow = {
		always = yes
	}
	potential = {
		hidden_trigger = {
			has_country_flag = BA_Setting_leader_rename_on
		}
	}
	effect = {
		hidden_effect = {
			remove_country_flag = BA_Setting_leader_rename_on
			remove_country_flag = BA_Setting_leader_rename_add_club_trait
			remove_country_flag = BA_Setting_leader_rename_add_skin_trait
		}
	}
}
