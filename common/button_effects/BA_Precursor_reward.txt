button_effect_BA_Precursor_reward_choice_base_bg_off = {
	allow = {
		always = yes
	}
	potential = {
		from = {
			NOT = {
				has_country_flag = BA_Precursor_reward_choice_base
			}
		}
	}
	effect = {
		custom_tooltip = BA_Precursor.reward.base
		hidden_effect = {
			from = {
				set_country_flag = BA_Precursor_reward_choice_base
				remove_country_flag = BA_Precursor_reward_choice_tech
				remove_country_flag = BA_Precursor_reward_choice_industry
				remove_country_flag = BA_Precursor_reward_choice_trade
			}
		}
	}
}
