##### Leaders
BA_leader_exempt_rename = {
	# scope: leader
	OR = {
		has_leader_flag = BA_event_leader
		has_leader_flag = BA_sp_leader
		is_same_value = owner.ruler
		has_leader_flag = star_oath_ruler
	}
}

BA_leader_has_same = {
	owner = {
		OR = {
			any_owned_leader = {
				has_leader_flag = $flag$
			}
			any_pool_leader = {
				has_leader_flag = $flag$
			}
		}
	}
}

BA_rename_special_check_1 = {
	has_total_skill >= 3
	species = {
		BA_has_halo_organic = yes
	}
	OR = {
		leader_class = commander
		leader_class = scientist
		leader_class = official
	}
	NOT = {
		has_leader_flag = BA_has_sp_trait_1
	}
}
