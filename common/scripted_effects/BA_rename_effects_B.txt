##### 系统初始化
#为所有领袖施加初始操作标记 country
BA_set_rename_flag = {
	every_owned_leader = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_target
	}
	every_pool_leader = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_target
	}
	every_envoy = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_target
	}
	# 重命名豁免
	every_owned_leader = {
		limit = {
			OR = {
				BA_leader_is_renamed = yes
				is_same_value = owner.ruler
				has_leader_flag = star_oath_ruler
			}
		}
		remove_leader_flag = BA_rename_target
	}
}

#遍历所有初始操作标记领袖，执行重命名 country
BA_iterate_rename_flag = {
	optimize_memory
	while = {
		limit = {
			any_owned_leader = {
				has_leader_flag = BA_rename_target
			}
		}
		random_owned_leader = {
			limit = {
				has_leader_flag = BA_rename_target
			}
			remove_leader_flag = BA_rename_target
			#检查命名标志
			BA_check_rename_flag = yes
			#按命名标志命名
			BA_rename_leader = yes
		}
	}
	while = {
		limit = {
			any_pool_leader = {
				has_leader_flag = BA_rename_target
			}
		}
		random_pool_leader = {
			limit = {
				has_leader_flag = BA_rename_target
			}
			remove_leader_flag = BA_rename_target
			#检查命名标志
			BA_check_rename_flag = yes
			#按命名标志命名
			BA_rename_leader = yes
		}
	}
	while = {
		limit = {
			any_envoy = {
				has_leader_flag = BA_rename_target
			}
		}
		random_envoy = {
			limit = {
				has_leader_flag = BA_rename_target
			}
			remove_leader_flag = BA_rename_target
			#特使专用处理
			BA_rename_envoy = yes
		}
	}
}

#生成事件领袖时，为重复的领袖更名 leader
BA_event_leader_replace = {
	optimize_memory
	set_leader_flag = $Leader_flag$
	owner = {
		every_owned_leader = {
			limit = {
				has_leader_flag = $Leader_flag$
				NOT = {
					is_same_value = prevprev
				}
			}
			remove_leader_flag = $Leader_flag$
			remove_leader_flag = BA_renamed
			remove_leader_flag = BA_has_sp_trait_1
			BA_remove_sp_trait = yes
			BA_check_rename_flag = yes
			BA_rename_leader = yes
		}
		every_pool_leader = {
			limit = {
				has_leader_flag = $Leader_flag$
			}
			remove_leader_flag = $Leader_flag$
			remove_leader_flag = BA_renamed
			remove_leader_flag = BA_has_sp_trait_1
			BA_remove_sp_trait = yes
			BA_check_rename_flag = yes
			BA_rename_leader = yes
		}
	}
}

#按设置执行重命名操作 leader
BA_rename_leader = {
	optimize_memory
	set_leader_flag = BA_renamed
	if = {
		limit = {
			has_global_flag = BA_leader_rename_activated
		}
		BA_set_name = yes
	}
	if = {
		limit = {
			has_global_flag = BA_leader_rename_add_club_trait
		}
		BA_add_club_trait = yes
	}
}

##### 修复操作
#为所有领袖施加修复操作标记 country
BA_set_rename_fix_flag = {
	every_owned_leader = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_fix_target
	}
	every_pool_leader = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_fix_target
	}
	every_envoy = {
		limit = {
			species = {
				BA_has_halo_organic = yes
			}
		}
		set_leader_flag = BA_rename_fix_target
	}
	# 重命名豁免
	every_owned_leader = {
		limit = {
			OR = {
				BA_leader_is_renamed = yes
				is_same_value = owner.ruler
				has_leader_flag = star_oath_ruler
			}
		}
		remove_leader_flag = BA_rename_target
	}
}

#遍历所有修复操作标记领袖，执行修复 country
BA_iterate_rename_fix_flag = {
	optimize_memory
	while = {
		limit = {
			any_owned_leader = {
				has_leader_flag = BA_rename_fix_target
			}
		}
		random_owned_leader = {
			limit = {
				has_leader_flag = BA_rename_fix_target
			}
			remove_leader_flag = BA_rename_fix_target
			#按命名标志命名
			BA_rename_leader = yes
			#移除特质并重新添加
			if = {
				limit = {
					OR = {
						has_global_flag = BA_leader_rename_add_club_trait
						has_global_flag = BA_leader_rename_add_skin_trait
					}
				}
				BA_remove_sp_trait = yes
				if = {
					limit = {
						has_global_flag = BA_leader_rename_add_club_trait
					}
					BA_add_club_trait = yes
				}
				if = {
					limit = {
						has_global_flag = BA_leader_rename_add_skin_trait
					}
					BA_change_skin_trait = yes
				}
			}
		}
	}
	while = {
		limit = {
			any_pool_leader = {
				has_leader_flag = BA_rename_fix_target
			}
		}
		random_pool_leader = {
			limit = {
				has_leader_flag = BA_rename_fix_target
			}
			remove_leader_flag = BA_rename_fix_target
			#按命名标志命名
			BA_rename_leader = yes
			#移除特质并重新添加
			if = {
				limit = {
					has_global_flag = BA_leader_rename_add_skin_trait
				}
				BA_remove_sp_trait = yes
				BA_change_skin_trait = yes
			}
		}
	}
	while = {
		limit = {
			any_envoy = {
				has_leader_flag = BA_rename_fix_target
			}
		}
		random_envoy = {
			limit = {
				has_leader_flag = BA_rename_fix_target
			}
			remove_leader_flag = BA_rename_fix_target
			#特使专用处理
			BA_rename_envoy = yes
		}
	}
}
