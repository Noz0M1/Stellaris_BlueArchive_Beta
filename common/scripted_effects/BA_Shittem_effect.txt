BA_Shittem_main_open = {
	#### 获取资源数量
	### 初始化
	remove_country_flag = BA_negative_pyroxenes
	remove_country_flag = BA_negative_keystone
	# 存量
	set_variable = {
		which = BA_var_stockpile_pyroxenes
		value = 0
	}
	set_variable = {
		which = BA_var_stockpile_keystone
		value = 0
	}
	set_variable = {
		which = BA_var_stockpile_energy
		value = 0
	}
	# 月度
	set_variable = {
		which = BA_var_income_pyroxenes
		value = 0
	}
	set_variable = {
		which = BA_var_income_keystone
		value = 0
	}
	### 载入
	# 存量
	export_resource_stockpile_to_variable = {
		resource = sr_pyroxenes
		variable = BA_var_stockpile_pyroxenes
	}
	export_resource_stockpile_to_variable = {
		resource = sr_keystone
		variable = BA_var_stockpile_keystone
	}
	export_resource_stockpile_to_variable = {
		resource = sr_eligma
		variable = BA_var_stockpile_energy
	}
	# 月度
	export_resource_income_to_variable = {
		resource = sr_pyroxenes
		variable = BA_var_income_pyroxenes
	}
	export_resource_income_to_variable = {
		resource = sr_keystone
		variable = BA_var_income_keystone
	}
	### 设置
	if = {
		limit = {
			check_variable = {
				which = BA_var_income_pyroxenes
				value < 0
			}
		}
		set_country_flag = BA_negative_pyroxenes
		multiply_variable = {
			which = BA_var_income_pyroxenes
			value = -1
		}
	}
	if = {
		limit = {
			check_variable = {
				which = BA_var_income_keystone
				value < 0
			}
		}
		set_country_flag = BA_negative_keystone
		multiply_variable = {
			which = BA_var_income_keystone
			value = -1
		}
	}
	#### 打开主界面
	random_list = {
		#事件提前生产0-24的随机数，通过该变量的值进行文本以及肖像的随机变化
		#肖像绑定值，文本绑定值，以此实现肖像与文本的绑定
		#序号为0的肖像与scripted loc当中的BlueArchivePulanaDesc0绑定，以此类推，Arona窗口同理
		1 = {
			set_variable_to_random_value = {
				which = BA_Arona_serial
				min = 0
				#max = 24
				max = 15
				rounded = yes
			}
			country_event = {
				id = BA_Shittem.1001
			}
		}
		1 = {
			set_variable_to_random_value = {
				which = BA_Pulana_serial
				min = 0
				#max = 24
				max = 13
				rounded = yes
			}
			country_event = {
				id = BA_Shittem.1002
			}
		}
	}
}

BA_club_resource_recount = {
	set_variable = {
		which = BA_club_cap_consumer_goods_cost
		value = BA_Club_limit
	}
	multiply_variable = {
		which = BA_club_cap_consumer_goods_cost
		value = 500
	}
	if = {
		limit = {
			check_variable = {
				which = BA_Club_limit
				value < 3
			}
		}
		set_variable = {
			which = BA_club_cap_pyroxenes_cost
			value = 0
		}
	}
	else = {
		set_variable = {
			which = BA_club_cap_pyroxenes_cost
			value = BA_Club_limit
		}
		subtract_variable = {
			which = BA_club_cap_pyroxenes_cost
			value = 2
		}
		multiply_variable = {
			which = BA_club_cap_pyroxenes_cost
			value = 100
		}
	}
	if = {
		limit = {
			check_variable = {
				which = BA_Club_limit
				value < 7
			}
		}
		set_variable = {
			which = BA_club_cap_keystone_cost
			value = 0
		}
	}
	else = {
		set_variable = {
			which = BA_club_cap_keystone_cost
			value = BA_Club_limit
		}
		subtract_variable = {
			which = BA_club_cap_keystone_cost

		multiply_variable = {
			which = BA_club_cap_keystone_cost
			value = 300
		}
	}
}

BA_club_set_modifier = {
	if = {
		limit = {
			check_variable = {
				which = BA_Club_check
				value >= 0
			}
		}
		remove_modifier = modifier_BA_club_overload_1
		remove_modifier = modifier_BA_club_overload_2
		remove_modifier = modifier_BA_club_overload_3
	}
	else_if = {
		limit = {
			check_variable = {
				which = BA_Club_check
				value >= -1
			}
		}
		add_modifier = {
			modifier = modifier_BA_club_overload_1
		}
		remove_modifier = modifier_BA_club_overload_2
		remove_modifier = modifier_BA_club_overload_3
	}
	else_if = {
		limit = {
			check_variable = {
				which = BA_Club_check
				value >= -2
			}
		}
		remove_modifier = modifier_BA_club_overload_1
		add_modifier = {
			modifier = modifier_BA_club_overload_2
		}
		remove_modifier = modifier_BA_club_overload_3
	}
	else = {
		remove_modifier = modifier_BA_club_overload_1
		remove_modifier = modifier_BA_club_overload_2
		add_modifier = {
			modifier = modifier_BA_club_overload_3
		}
	}
}

	# this = starbase
	fleet = {
		if = {
			limit = {
				any_owned_ship = {
					is_ship_size = BA_defence_cannon
				}
			}
			random_owned_ship = {
				limit = {
					is_ship_size = BA_defence_cannon
				}
				save_event_target_as = target_cannon
			}
		}
		else = {
			owner = {
				random_owned_design = {
					limit = {
						is_ship_size = BA_defence_cannon
					}
					save_event_target_as = target_cannon
				}
			}
		}
		create_ship = {
			name = random
			design = event_target:target_cannon
			graphical_culture = owner
		}
	}
}
