# 先驱者报酬资源：调用前必须已将 BA_Precursor_unity_spent 设为本档累计消耗阈值。
# 资源数值均以该阈值为 multiplier，不按档位另行拆分。
# 根据奖励界面当前选择派发资源。
BA_Precursor_reward_resource = {
	if = {
		limit = {
			has_country_flag = BA_Precursor_reward_choice_base
		}
		add_resource = {
			energy = 2
			food = 1
			consumer_goods = 1
			multiplier = BA_Precursor_unity_spent
		}
	}
	else_if = {
		limit = {
			has_country_flag = BA_Precursor_reward_choice_tech
		}
		add_resource = {
			physics_research = 0.5
			society_research = 0.5
			engineering_research = 0.5
			multiplier = BA_Precursor_unity_spent
		}
	}
	else_if = {
		limit = {
			has_country_flag = BA_Precursor_reward_choice_industry
		}
		add_resource = {
			minerals = 2
			alloys = 0.5
			multiplier = BA_Precursor_unity_spent
		}
	}
	else_if = {
		limit = {
			has_country_flag = BA_Precursor_reward_choice_trade
		}
		add_resource = {
			trade = 3
			multiplier = BA_Precursor_unity_spent
		}
	}
}
