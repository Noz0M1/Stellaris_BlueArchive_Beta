### 功能1：赋予身份：赋予领袖一个身份flag（排除豁免）
BA_rename_base_identity = {
	# scope: leader
	if = {
		limit = {
			BA_rename_identity_target_check = yes
		}
		set_leader_flag = BA_rename_identity
		BA_rename_effect_identity = yes
	}
}

		}
		BA_rename_effect_portrait = yes
	}
}

### 功能3：社团特质
BA_rename_base_club_trait = {
	# scope: leader
	if = {
		limit = {
			has_global_flag = BA_leader_rename_add_club_trait
			has_leader_flag = BA_rename_identity
		}
		if = {
			limit = {
				has_leader_flag = BA_rename_club_trait
			}
			remove_leader_flag = BA_rename_club_trait
			BA_rename_support_club_trait_remove = yes
		}
		set_leader_flag = BA_rename_club_trait
		BA_rename_effect_club_trait = yes
	}
}

### 功能4：个人特质
BA_rename_base_skin_trait = {
	# scope: leader
	if = {
		limit = {
			has_global_flag = BA_leader_rename_add_skin_trait
			has_leader_flag = BA_rename_identity
			has_total_skill >= 3
		}
		if = {
			limit = {
				has_leader_flag = BA_rename_skin_trait
			}
			remove_leader_flag = BA_rename_skin_trait
			BA_rename_support_skin_trait_remove = yes
		}
		set_leader_flag = BA_rename_skin_trait
		BA_rename_effect_skin_trait = yes
	}
}
