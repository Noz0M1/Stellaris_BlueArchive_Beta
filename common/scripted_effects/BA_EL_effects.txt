BA_Aris_EL = {
	create_leader = {
		class = $class$
		species = owner_main_species
		gender = female
		name = NAME_Aris
		skill = 8
		set_age = 1
		immortal = yes
		event_leader = yes
		traits = {
			trait = leader_trait_Aris_EL1
		}
		effect = {
			BA_Halo_for_special_leader = yes
			set_leader_flag = leader_death_events_blocked
			set_leader_flag = BA_elite_leader
			set_leader_flag = BA_rename_identity
			set_leader_flag = leader_BA_Aris
			save_global_event_target_as = BA_EL_live_Aris_@owner
			owner = {
				BA_rename_base_change_dupl_country = {
					NAME = Aris
				}
			}
			change_leader_portrait = BA_leader_Aris
			owner = {
				set_timed_country_flag = {
					flag = BA_Aris_cooldown
					days = 360
				}
			}
		}
	}
	inline_script = {
		script = Leader/EL_clear_preview
		Leader = Aris
	}
}

# Country scope. Change the existing elite Aris in place so that her level,
# experience, traits and per-leader state survive a role change.
BA_Aris_EL_change_class = {
	if = {
		limit = {
			any_owned_leader = {
				BA_is_elite_leader = yes
				has_leader_flag = leader_BA_Aris
			}
		}
		every_owned_leader = {
			limit = {
				BA_is_elite_leader = yes
				has_leader_flag = leader_BA_Aris
			}
			hidden_effect = {
				unassign_leader = this
			}
			change_leader_class = $class$
		}
		set_timed_country_flag = {
			flag = BA_Aris_cooldown
			days = 360
		}
	}
}

# Country scope. Killing Aris without a notification routes through
# on_leader_death_no_notify and the standard EL absence capture.
BA_Aris_EL_to_matrix = {
	every_owned_leader = {
		limit = {
			BA_is_elite_leader = yes
			has_leader_flag = leader_BA_Aris
		}
		kill_leader = {
			show_notification = no
		}
	}
	add_modifier = {
		modifier = modifier_BA_Aris_in_matrix
	}
}
