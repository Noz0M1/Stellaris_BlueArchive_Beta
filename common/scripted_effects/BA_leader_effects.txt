## 定制领袖
BA_governor_Serina_PR = {
	create_leader = {
		class = official
		species = event_target:BA_Precursor_country.owner_main_species
		gender = female
		name = NAME_Serina
		skill = 3
		set_age = 16
		immortal = yes
		event_leader = yes
		traits = {
			trait = leader_trait_customize
			trait = leader_trait_custom_Serina_SP
		}
	}
	last_created_leader = {
		BA_Halo_for_special_leader = yes
		set_leader_flag = BA_event_leader
		set_age = 16
		change_leader_portrait = BA_leader_Serina
		set_leader_flag = BA_skin_XMS
		set_leader_flag = BA_Serina_PR
		set_leader_flag = BA_rename_identity
		set_leader_flag = leader_BA_Serina
		owner = {
			BA_rename_base_change_dupl_country = {
				NAME = Serina
			}
		}
	}
}
