option = {
	name = Xuanwu_promenade_trade.consumer_goods
	response_text = Xuanwu_promenade_trade.$CHARA$RE.1
	trigger = {
	}
	allow = {
		resource_stockpile_compare = {
			resource = trade
			value >= 50000
		}
	}
	add_resource = {
		trade = -50000
	}
	add_resource = {
		consumer_goods = 5000
		multiplier = value:num_BA_club_trade_mult
	}
}

#三级资源
option = {
	name = Xuanwu_promenade_trade.volatile_motes
	response_text = Xuanwu_promenade_trade.$CHARA$RE.1
	trigger = {
		OR = {
			has_technology = tech_volatile_motes
			has_technology = tech_mine_volatile_motes
			has_technology = tech_BA_angel24_rare_produce
		}
	}
	allow = {
		custom_tooltip_fail = {
			text = Xuanwu_promenade_trade.noflow
			NOT = {
				resource_income_compare = {
					resource = volatile_motes
					value = 0
				}
			}
		}
		custom_tooltip_fail = {
			text = BA_club_favor_fail.50
			check_variable = {
				which = Xuanwu_promenade
				value >= 50
			}
		}
		resource_stockpile_compare = {
			resource = trade
			value >= 50000
		}
	}
	add_resource = {
		trade = -50000
	}
	add_resource = {
		volatile_motes = 1000
		multiplier = value:num_BA_club_trade_mult
	}
}
